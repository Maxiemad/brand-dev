📜 Enhanced JavaScript Script

Complete enhanced script with scroll timing and multiple attempts:

<script> // Enhanced redirect function with scroll handling function redirectToPDFDownloadEnhanced() { // First, navigate to the page window.location.href = "https://planning.gotoretreats.com/#pdf-download"; // Wait for page to load, then handle scrolling setTimeout(() => { scrollToPDFSection(); }, 500); // Wait 500ms for page to load } // Function to scroll to PDF section with multiple attempts function scrollToPDFSection() { // First scroll attempt at 300ms setTimeout(() => { const pdfSection = document.getElementById('pdf-download') || document.querySelector('[data-section="pdf-download"]') || document.querySelector('.pdf-download-section'); if (pdfSection) { pdfSection.scrollIntoView({ behavior: 'smooth', block: 'start' }); } }, 300); // Second scroll attempt at 800ms (backup) setTimeout(() => { const pdfSection = document.getElementById('pdf-download') || document.querySelector('[data-section="pdf-download"]') || document.querySelector('.pdf-download-section'); if (pdfSection) { pdfSection.scrollIntoView({ behavior: 'smooth', block: 'start' }); } }, 800); } </script>
